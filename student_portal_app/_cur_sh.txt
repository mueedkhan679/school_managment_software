   6: /// A single shimmering rectangle used to compose skeleton layouts.
   7: class ShimmerPlaceholder extends StatelessWidget {
   8:   const ShimmerPlaceholder({
   9:     super.key,
  10:     this.width = double.infinity,
  11:     this.height = 14,
  12:     this.radius = 8,
  13:   });
  14: 
  15:   final double? width;
  16:   final double height;
  17:   final double radius;
  18: 
  19:   @override
  20:   Widget build(BuildContext context) {
  21:     final isDark = Theme.of(context).brightness == Brightness.dark;
  22:     return Shimmer.fromColors(
  23:       baseColor: isDark ? const Color(0xFF1C2A47) : const Color(0xFFE3EAF5),
  24:       highlightColor:
  25:           isDark ? const Color(0xFF283A60) : const Color(0xFFF8FBFF),
  26:       child: Container(
  27:         width: width,
  28:         height: height,
  29:         decoration: BoxDecoration(
  30:           color: Colors.white,
  31:           borderRadius: BorderRadius.circular(radius),
  32:         ),
  33:       ),
  34:     );
  35:   }
  36: }

  38: /// Animated skeleton list for student rosters, fee schedules and history
  39: /// ledgers while data is being fetched.
  40: class SkeletonList extends StatelessWidget {
  41:   const SkeletonList({
  42:     super.key,
  43:     this.itemCount = 6,
  44:     this.shrinkWrap = false,
  45:     this.padding = const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
  46:     this.horizontalPadding = 16,
  47:   });
  48: 
  49:   final int itemCount;
  50:   final bool shrinkWrap;
  51:   final EdgeInsetsGeometry padding;
  52:   final double horizontalPadding;
  53: 
  54:   @override
  55:   Widget build(BuildContext context) {
  56:     return ListView.separated(
  57:       shrinkWrap: shrinkWrap,
  58:       padding: padding,
  59:       physics:
  60:           shrinkWrap ? const NeverScrollableScrollPhysics() : null,
  61:       itemCount: itemCount,
  62:       separatorBuilder: (_, __) => const SizedBox(height: 10),
  63:       itemBuilder: (_, __) => Container(
  64:         padding: const EdgeInsets.all(14),
  65:         decoration: BoxDecoration(
  66:           color: Theme.of(context).colorScheme.surface,
  67:           borderRadius: BorderRadius.circular(16),
  68:           boxShadow: SchoolPalette.softShadow,
  69:         ),
  70:         child: const Row(
  71:           children: [
  72:             ShimmerPlaceholder(width: 44, height: 44, radius: 22),
  73:             SizedBox(width: 14),
  74:             Expanded(
  75:               child: Column(
  76:                 crossAxisAlignment: CrossAxisAlignment.start,
  77:                 children: [
  78:                   ShimmerPlaceholder(
  79:                     width: double.infinity,
  80:                     height: 13,
  81:                   ),
  82:                   SizedBox(height: 8),
  83:                   ShimmerPlaceholder(width: 130, height: 10, radius: 6),
  84:                 ],

  86:             ),

  92:     );
