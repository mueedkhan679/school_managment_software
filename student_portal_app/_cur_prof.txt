  58:   static String? _parsePhotoUrl(dynamic url) {
  59:     if (url == null) return null;
  60:     final String raw = url.toString().trim();
  61:     if (raw.isEmpty) return null;
  62: 
  63:     // Already an absolute URL â€“ use it as-is.
  64:     if (raw.startsWith('http://') || raw.startsWith('https://')) {
  65:       return raw;
  66:     }
  67: 
  68:     // Relative backend media path (e.g. "/media/student_photos/x.jpg" or
  69:     // "media/student_photos/x.jpg") â€“ resolve it against the API origin so
  70:     // uploaded student photos always render.
  71:     String base = ApiService.baseUrl;
  72:     if (base.endsWith('/')) {
  73:       base = base.substring(0, base.length - 1);
  74:     }
  75:     return raw.startsWith('/') ? '$base$raw' : '$base/$raw';
  76:   }
