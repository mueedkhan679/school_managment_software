 247:       itemCount: schedule.length,

 251: 
 252:         return Card(

 259:                 isPaid ? Icons.check_circle_rounded : Icons.schedule_rounded,
 260:                 color: isPaid ? Colors.green : Colors.orange,
 261:               ),
 262:             ),
 263:             // Title: month name (e.g. "January 2026")
