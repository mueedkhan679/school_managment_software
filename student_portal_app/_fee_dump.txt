229:   Widget _buildScheduleTab(List<MonthlyScheduleItem> schedule, bool isLoading, ThemeData theme) {
230:     if (isLoading) {
231:       return const Center(child: CircularProgressIndicator());
232:     }
233:     if (schedule.isEmpty) {
234:       return const Center(child: Text('No fee schedule available'));
235:     }
236:     return ListView.builder(
237:       padding: const EdgeInsets.all(16),
238:       itemCount: schedule.length,
239:       itemBuilder: (context, index) {
240:         final item = schedule[index];
241:         final isPaid = item.isPaid;
242: 
243:         return Card(
244:           margin: const EdgeInsets.only(bottom: 10),
245:           shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
246:           child: ListTile(
247:             leading: CircleAvatar(
248:               backgroundColor: isPaid ? Colors.green.withValues(alpha: 0.15) : Colors.orange.withValues(alpha: 0.15),
249:               child: Icon(
250:                 isPaid ? Icons.check_circle_rounded : Icons.schedule_rounded,
251:                 color: isPaid ? Colors.green : Colors.orange,
252:               ),
253:             ),
254:             // Title: month name (e.g. "January 2026")
255:             title: Text(
256:               item.monthName,
257:               maxLines: 1,
258:               overflow: TextOverflow.ellipsis,
259:               style: const TextStyle(fontWeight: FontWeight.bold),
260:             ),
261:             // Subtitle: fee amount (plus payment date when available)
262:             subtitle: Text(
263:               (isPaid && (item.paymentDate?.isNotEmpty ?? false))
264:                   ? '${Format.rupees(item.amount)}  â€¢  Paid ${item.paymentDate}'
265:                   : Format.rupees(item.amount),
266:               style: const TextStyle(fontSize: 13),
267:             ),
268:             // Trailing: green PAID badge with check icon / red PENDING badge
269:             trailing: Container(
270:               padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 6),
271:               decoration: BoxDecoration(
272:                 color: isPaid ? Colors.green : Colors.red,
273:                 borderRadius: BorderRadius.circular(20),
274:               ),
275:               child: Row(
276:                 mainAxisSize: MainAxisSize.min,
